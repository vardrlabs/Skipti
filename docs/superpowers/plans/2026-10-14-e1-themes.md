# E1 (D-212): the theme engine and Appearance, Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Three themes (Aurora, the new default; Norse, always dark; Vardr Labs) by light and dark, a text size of -2 to +3 steps from the phone's own, and an Appearance page to choose them, on both shells, every colour pair measured.

**Architecture:** Every colour token gets five columns (Aurora light, Aurora dark, Norse, Vardr Labs light, Vardr Labs dark). iOS resolves a column from the window's trait collection (the style plus a new `ThemeTrait`, as Night rode `NightTrait`); Android's `Tok(dark)` resolves it from snapshot state (`Appearance.theme`, as it read `Appearance.night`). Faces follow the theme through one observable state on iOS (`AppearanceState.shared`, read inside `TypeRole`, so every body that uses a role re-evaluates on a change) and through snapshot state on Android. Text size is a window trait override on iOS and a root `LocalDensity` on Android. No screen-by-screen rewrite: screens keep reading `Tokens.x`, `TypeRole.x(`, `Tok.x(dark)`, `Type.x(`.

**Tech Stack:** SwiftUI + UIKit traits (iOS 17 target), Observation; Jetpack Compose; Python 3 (the palette derivation and the licence check); no new code dependency. One new asset: Cormorant SC (OFL 1.1, downloaded with the founder's yes, 2026-10-14).

**Spec:** `docs/superpowers/specs/2026-10-14-e1-themes-design.md` (approved 2026-10-14). Read it first. Visual reference: the Look Book v2.1, `.drafts/lookbook/skipti-lookbook.html`.

**Branch:** `e1-themes` from main `1f17456`. D-number **D-212**. Drafts in `.drafts/d212/`.

## Text size

Steps -2...3, 0 the phone's own size. iOS: the phone's `UIContentSizeCategory` moved along this ladder and clamped at both ends: `.extraSmall, .small, .medium, .large, .extraLarge, .extraExtraLarge, .extraExtraExtraLarge, .accessibilityMedium, .accessibilityLarge, .accessibilityExtraLarge, .accessibilityExtraExtraLarge, .accessibilityExtraExtraExtraLarge` (an unknown phone category counts as `.large`). Android: the phone's `fontScale` multiplied by the iOS body-size ratio of the same step from Large (body 15/16/17/19/21/23 pt), clamped to 0.75...2.0: **-2: 0.88, -1: 0.94, 0: 1.0, +1: 1.12, +2: 1.24, +3: 1.35**.

```swift
import UIKit
import XCTest
@testable import Skipti

/// D-212 (ADR-0041): three themes by light and dark, the text size, and the
/// floors every pair must clear in every set. Android's twin is AppearanceTest.
final class AppearanceTests: XCTestCase {
    private func traits(_ set: ThemeSet) -> UITraitCollection {
        UITraitCollection { t in
            t.userInterfaceStyle = set.isDark ? .dark : .light
            t[ThemeTrait.self] = set.theme
        }
    }

    private func hex(_ color: UIColor) -> String {
        var r: CGFloat = 0, g: CGFloat = 0, b: CGFloat = 0, a: CGFloat = 0
        color.getRed(&r, green: &g, blue: &b, alpha: &a)
        return String(format: "#%02X%02X%02X", Int(round(r * 255)), Int(round(g * 255)), Int(round(b * 255)))
    }

    private func luminance(_ color: UIColor) -> Double {
        var r: CGFloat = 0, g: CGFloat = 0, b: CGFloat = 0, a: CGFloat = 0
        color.getRed(&r, green: &g, blue: &b, alpha: &a)
        func channel(_ c: CGFloat) -> Double {
            let v = Double(c)
            return v <= 0.03928 ? v / 12.92 : pow((v + 0.055) / 1.055, 2.4)
        }
        return 0.2126 * channel(r) + 0.7152 * channel(g) + 0.0722 * channel(b)
    }

    private func contrast(_ a: UIColor, _ b: UIColor) -> Double {
        let (hi, lo) = (max(luminance(a), luminance(b)), min(luminance(a), luminance(b)))
        return (hi + 0.05) / (lo + 0.05)
    }
```

```swift
    static func apply(_ state: AppearanceState) {

            guard let windowScene = scene as? UIWindowScene else { continue }

    /// itself an override and resolved as light). Step 0 removes the text
    /// size override the same way, so the phone's own size keeps following.
    @MainActor
    static func apply(_ state: AppearanceState, to window: UIWindow) {
        let style = state.mode.interfaceStyle(theme: state.theme)
        if style == .unspecified {
            window.traitOverrides.remove(UITraitUserInterfaceStyle.self)
        } else {
            window.traitOverrides.userInterfaceStyle = style
        }
        window.traitOverrides[ThemeTrait.self] = state.theme
        if state.textStep == 0 {
            window.traitOverrides.remove(UITraitPreferredContentSizeCategory.self)
        } else {

/// Applies the stored appearance when the root appears, and again whenever a
/// setting changes (the Appearance page writes the same keys) or the phone's
/// own text size changes.

    @AppStorage(AppTheme.storageKey) private var themeRaw = AppTheme.aurora.rawValue
    @AppStorage(AppearanceMode.storageKey) private var modeRaw = AppearanceMode.system.rawValue
    @AppStorage(TextSize.storageKey) private var textStep = 0

    func body(content: Content) -> some View {
        content
            .environment(\.appTheme, AppTheme.parse(themeRaw))
            .tint(Tokens.accent)
            .onAppear { sync() }
            .onChange(of: themeRaw) { sync() }
            .onChange(of: modeRaw) { sync() }
            .onChange(of: textStep) { sync() }
            .onReceive(NotificationCenter.default.publisher(for: UIContentSizeCategory.didChangeNotification)) { _ in
                sync()
            }
    }

    @MainActor
    private func sync() {
        let state = AppearanceState.shared
        state.theme = AppTheme.parse(themeRaw)
        state.mode = AppearanceMode.parse(modeRaw)
        state.textStep = TextSize.parse(textStep)
        state.phoneCategory = UIApplication.shared.preferredContentSizeCategory
        Appearance.apply(state)
    }
}
```

```kotlin
object TextSize {
    const val MIN = -2
    const val MAX = 3
    private val FACTORS = mapOf(-2 to 0.88f, -1 to 0.94f, 0 to 1.0f, 1 to 1.12f, 2 to 1.24f, 3 to 1.35f)

    /** A stored step outside the range (or none) is the phone's own size. */
    fun parse(raw: Int?): Int = if (raw != null && raw in MIN..MAX) raw else 0

    fun factor(step: Int): Float = FACTORS[parse(step)] ?: 1.0f

    fun fontScale(phone: Float, step: Int): Float = (phone * factor(step)).coerceIn(0.75f, 2.0f)

        step == -1 -> "1 step smaller"
```

```swift
    static func base(_ face: Face, size: CGFloat, weight: CGFloat, tabular: Bool) -> UIFont {
        switch face {
        case .grotesk:
            return uiFont(grotesk, size: size, weight: weight, tabular: tabular)

            guard tabular else { return font }
            let feature: [UIFontDescriptor.FeatureKey: Int] = [.type: kNumberSpacingType, .selector: kMonospacedNumbersSelector]
            return UIFont(descriptor: font.fontDescriptor.addingAttributes([.featureSettings: [feature]]), size: size)
        case .cormorant:
            return uiFont(weight >= 650 ? cormorantBold : cormorantSemiBold, size: size, weight: nil, tabular: tabular)
        case .plex:
            return uiFont(weight >= 500 ? plexMedium : plexRegular, size: size, weight: nil, tabular: tabular)
        }

    /// text-size step), not the window's trait alone: `AppearanceState` is
    /// observable, so a body that calls a role re-evaluates when it changes.
    static func scaledSize(_ size: CGFloat, relativeTo style: UIFont.TextStyle, category: UIContentSizeCategory) -> CGFloat {
        UIFontMetrics(forTextStyle: style).scaledValue(for: size, compatibleWith: UITraitCollection(preferredContentSizeCategory: category))
```
